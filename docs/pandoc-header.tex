% ---------------------------------------------------------------------------
% XeLaTeX header for Draft_Dissertation.pdf.  Used by scripts/build_pdf.sh.
%
% Depends only on packages present in the TeX Live "basic" scheme (pifont,
% etoolbox), so the PDF builds on a stock installation with no tlmgr step.
%
% Three jobs:
%   1. Supply the two glyphs Latin Modern lacks -- U+2717 BALLOT X and U+2713
%      CHECK MARK -- used in the validation and ground-truth-provenance tables.
%   2. Set tables at \footnotesize and tighten the inter-column padding. Pandoc
%      sizes each p{} column as (\linewidth - 2*ncols*\tabcolsep) * fraction, so
%      padding is subtracted from the content width before the proportions are
%      applied; at the default 6pt the nine-column machine table (Table 4) spends
%      ~108pt of a ~452pt text block on gutters alone, which pushes its narrow
%      cells past the margin.
%   3. Loosen the line-breaking tolerance. The prose is dense with \texttt{} runs
%      and inline maths, and the default badness threshold yields avoidable
%      overfull lines.
%
% Verified: 0 overfull boxes, 0 missing characters.
% ---------------------------------------------------------------------------

\usepackage{pifont}

% Latin Modern carries every glyph the document uses except these two dingbats
% (U+2717 BALLOT X and U+2713 CHECK MARK), which appear in the validation and
% ground-truth-provenance tables. Declaring the characters active and binding
% them to pifont is what the newunicodechar package does internally; doing it
% directly avoids a dependency absent from the TeX Live "basic" scheme, and is
% preferable to switching the whole document to a broad-coverage sans face for
% the sake of two characters.
\catcode`✗=\active
\def✗{\ding{55}}
\catcode`✓=\active
\def✓{\ding{51}}

% Latin Modern's *text* face also lacks the relational operators, though its math
% face carries them. Route them through \ensuremath so they typeset correctly
% wherever they appear in prose or in a table cell.
\catcode`≤=\active
\def≤{\ensuremath{\leq}}
\catcode`≥=\active
\def≥{\ensuremath{\geq}}
\catcode`≈=\active
\def≈{\ensuremath{\approx}}

% Latin Modern does carry the glyphs below, but declaring them costs nothing and
% makes the build independent of which Latin Modern revision is installed.
\newcommand{\aeminus}{\textminus}
\newcommand{\aearrow}{\textrightarrow}

\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\footnotesize}
\AtBeginEnvironment{tabular}{\footnotesize}
\setlength{\tabcolsep}{3pt}

\tolerance=2000
\emergencystretch=3em

% Allow breaks inside long monospaced identifiers (file names, CLI flags) so a
% wide \texttt{} run cannot single-handedly push a line past the margin.
\renewcommand{\texttt}[1]{{\ttfamily\hyphenchar\font=`\-\relax #1}}
